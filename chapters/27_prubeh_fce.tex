\section{Průběh funkce}
\begin{pozn}
    Musíme umět nakreslit graf a popsat základní vlastnosti lineární a
    kvadratické funkce, nepřímé úměrnosti, lineární lomené, mocninné, exponenciální,
    logaritmické, goniometrické a cyklometrické funkce.
\end{pozn}

\begin{veta}
Nechť funkce $f$ má na intervalu $(a,b),a,b\in \mathbb R^*$ derivaci. Je-li pro
všechna $x$~z~intervalu $(a,b)$
\begin{enumerate}[$i.$]
\item $f^\prime(x)>0,$ pak je $f$ rostoucí;
\item $f^\prime(x)\geq 0,$ pak je $f$ neklesající;
\item $f^\prime(x)<0,$ pak je $f$ klesající;
\item $f^\prime(x) \leq 0,$ pak je $f$ nerostoucí;
\item $f^\prime(x)=0,$ pak je $f$ konstantní
\end{enumerate}
na intervalu $(a,b).$
\end{veta}

\begin{priklad}
Určete monotónnost a lokální extrémy funkce $f(x)=\frac{x^4}{4-x^2}$.
\end{priklad}

\begin{reseni}
Spočteme první derivaci funkce $f(x)$. Je
$$
    f^\prime (x) = \frac{4x^3 \left( 4-x^2 \right ) - x^4 \left( -2x \right )  }{\left( 4-x^2 \right )^2 } = \frac{2x^3 \left( 8-x^2 \right ) }{\left( 4-x^2 \right )^2 } = \frac{2x^3 \left( 2 \sqrt{2} -x  \right ) \left( 2 \sqrt{2} + x \right )  }{\left( 2-x \right )^2 \left( 2+x \right )^2 }.
$$
Dále znaménka funkce $f^\prime (x)$ a zaneseme je na číselnou osu.

\begin{definition}
Funkce $f$ má v bodě $x_0$ \textbf{lokální minimum} (resp. \textbf{maximum}),
jestliže existuje okolí $\mathscr O(x_0)$ takové, že $\forall x \in \mathscr O(x_0)$ platí
\begin{align*}
    f(x)\geq f(x_0), & & \textrm{resp. } f(x)\leq f(x_0).
\end{align*}
Lokální minimum (resp. maximum) je \textbf{ostré}, jestliže
\begin{align*}
    f(x)> f(x_0), & & \textrm{resp. } f(x)<f(x_0).
\end{align*}
\end{definition}

\begin{definition}
Bod $x_0\in \mathcal D(f)$ takový, že $f^\prime(x_0)=0$ je \textbf{stacionární bod}.
\end{definition}

\begin{veta}
Nechť $f$ má v bodě $x_0$ lokální extrém. Pak buď $f^\prime(x_0)=0,$ nebo
$f^\prime(x_0)$ neexistuje.
\end{veta}

\begin{definition}\label{konvkonk}
Funkce $f$ je \textbf{konvexní} (resp. \textbf{konkávní}) v bodě $x_0\in \mathcal D(f),$
jestliže existuje okolí $\mathscr O(x_0)$ takové, že pro všechna $x\in\mathscr O(x_0)$
platí
\begin{align*}
    f(x)\geq g(x), & & \textrm{resp. } f(x)\leq g(x),
\end{align*}
kde $g(x)$ jsou funkční hodnoty na tečně v bodě $(x_0,f(x_0)).$

Funkce je konvexní (resp. konkávní) na intervalu, jestliže je konvexní (resp.
konkávní) v~každém jeho bodě.
\end{definition}

\begin{definition}
Funkce $f$ má v bodě $x_0$ \textbf{inflexi}, jestliže existuje $f^\prime(x_0) \in
\mathbb R$ a $f$ je v nějakém levém okolí bodu $x_0$ konvexní a v nějakém pravém okolí
tohoto bodu konkávní, resp. naopak. Má-li funkce $f$ v bodě $x_0$ inflexi, pak bod
$(x_0, f(x_0))$ nazýváme \textbf{inflexním bodem} funkce $f$.
\end{definition}

\begin{veta}
Nechť má funkce $f$ v intervalu $(a,b)$ druhou derivaci. Je-li
\begin{enumerate}[$i.$]
\item $f'' (x)>0$ pro všechna $x \in (a,b)$, pak je $f$
konvexní na $(a,b)$;
\item $f''(x)<0$ pro všechna $x \in (a,b)$, pak je $f$
konkávní na $(a,b)$;
\item $f''(x)=0$ v nějakém bodě $x_0\in(a,b)$ a dále je $f''$
kladná v nějakém levém okolí bodu $x_0$ a záporná v nějakém pravém okolí bodu $x_0$,
resp. naopak, pak má $f$ v $x_0$ inflexi.
\end{enumerate}
\end{veta}

\begin{definition}
Přímka $x=x_0,x_0\in \mathbb R$ se nazývá \textbf{asymptota bez směrnice}
grafu funkce $f$, jestliže alespoň jedna jednostranná limita funkce $f$
v bodě $x_0$ je nevlastní, tj.
\begin{align*}
    \lim_{x\to x_0^+} f(x) = \pm\infty, & & \lim_{x\to x_0^-} f(x)=\pm\infty.
\end{align*}
\end{definition}

\begin{definition}
Přímka $y=ax+b, a,b\in \mathbb R$ se nazývá \textbf{asymptota se směrnicí} grafu
funkce $f$ v $+\infty$ (resp. $-\infty$), jestliže
\begin{align*}
    \lim_{x\to\infty}(f(x)-(ax+b))=0, & & \textrm{resp. }\lim_{x\to-\infty}(f(x)-(ax+b))=0.
\end{align*}
\end{definition}

\begin{veta}
    Přímka $y=ax+b$ je asymptota se směrnicí v $\pm\infty$ právě tehdy, když
    \begin{align*}
        \lim_{x\to\pm\infty}\frac{f(x)}{x}=a, a\in \mathbb R, & & \lim_{x\to\pm\infty}(f(x)-ax)=b,b \in \mathbb R.
    \end{align*}
\end{veta}

\begin{priklad}
Najděte asymptoty grafu funkce $f(x)=\frac{x^2-2x}{x-1}.$
\end{priklad}

\begin{reseni}
Asymptoty bez směrnice nalezneme počítáním jednostranných limit v bodech nespojitosti. Funkce $f(x)$ má jediný
bod nespojitosti, a to $x = 1$. Pak je
$$
\lim_{x \to 1 +} \frac{x^2-2x}{x-1} = \frac{1-2}{0+} = \frac{-1}{0+} = -\infty.
$$
Obdobně $\lim_{x \to 1 -}f(x) = \infty$. U asymptot se směrnicí hledáme přímku $y=ax+b$,
kde $a=\lim_{x\to \pm \infty}\frac{f(x)}{x}$ a
$b=\lim_{x\to \pm \infty}(f(x)-ax).$ Je tedy
$$
    a = \lim_{x\to \pm \infty} \frac{x-2}{x-1} \cdot \frac{\frac{1}{x}}{\frac{1}{x}} = \lim_{x\to \pm \infty} \frac{1 - \frac{2}{x}}{1 - \frac{1}{x}} = 1.
$$
Potom
$$
    b= \lim_{x\to \pm \infty} \frac{x^2 - 2x - x(x-1)}{x-1} = \lim_{x\to \pm \infty} \frac{-x}{x-1} \cdot \frac{\frac{1}{x}}{\frac{1}{x}} = \lim_{x\to \pm \infty} \frac{-1}{1 - \frac{1}{x}} = -1.
$$
Asymptota se směrnicí do $\pm \infty$ je tedy přímka $x-1$.
\end{reseni}

\begin{pozn}
    Při určování průběhu funkce musíme
    \begin{enumerate}[$i.$]
    \item vyšetřit $\mathcal D(f)$, body nespojitosti, nulové body, znaménka funkce, popř.
    sudost/lichost, periodičnost;
   	\item určit intervaly monotonie, lokální extrémy;
   	\item určit intervaly konvexnosti/konkávnosti, inflexní body;
   	\item určit asymptoty bez směrnice a se směrnicí;
   	\item načrtnout graf funkce.
    \end{enumerate}
\end{pozn}

\begin{priklad}
Vyšetřete průběh funkce $f(x)=\frac{(x-1)^3}{(x-2)^2}$.
\end{priklad}

\begin{reseni}
\begin{enumerate} [$i.$]
	\item Definiční obor funkce je $\mathbb R \smallsetminus \left \{ 2 \right \} $. Funkce je nulová v bodě
	$x=1$. Funkce je záporná na intervalu $\left (-\infty, 1 \right ) $ a kladná na intervalech
	$\left( 1,2 \right )\cup \left( 2,\infty \right )  $. Funkce není ani sudá/lichá, ani periodická.
	\item Intervaly monotonie dostaneme z první derivace. Ta je rovna
	\begin{align*}
  f^\prime(x) &= \frac{3(x-1)^2(x-2)^2 - (x-1)^3(x-2)}{(x-2)^4}\\
  &=\frac{(x-1)^2(x-2)\left [ 3(x-2)-2(x-1) \right ] }{(x-2)^4}= \frac{(x-1)^2(x-4)}{(x-2)^3}.
	\end{align*}
	Odsud vidíme, že funkce je rostoucí na intervalech $\left( -\infty, 1 \right ) $, $\left( 1, 2 \right ) $
	a $\left( 4,\infty \right ) $, klesající na intervalu $\left( 2,4 \right ) $, v bodě
	$x=2$ ja lokální maximum a v bodě $x=4$ je lokální minimum.
	\item Intervaly konvexnosti zjistíme z druhé derivace. Je
	\begin{align*}
		f^{\prime\prime}(x) &= \frac{\left( 2(x-1)(x-4) + (x-1)^2 \right )(x-2)^3-3(x-1)^2(x-4)(x-2)^2}{(x-2)^6}\\
		&= \frac{3(x-1)\left [ (x-2)(x-3) - (x-1)(x-4) \right ] }{(x-2)^4}\\
		&= \frac{6(x-1)}{(x-2)^4}.
	\end{align*}
	Odsud vidíme, že na intervalu $\left( -\infty, 1 \right ) $ je funkce konkávní a na intervalech
	$\left( 1,2 \right ) $ a~$\left( 2,\infty \right ) $ je funkce konvexní. V bodě $x=1$ má inflexní bod.
	\item Asymptoty bez směrnice jsou
	\begin{align*}
	 \lim_{x \to 2+} \frac{(x-1)^3}{(x-2)^2} = \infty, && \lim_{x \to 2-}\frac{(x-1)^3}{(x-2)^2} = \infty.
	\end{align*}
	Asymptota se směrnicí $ax+b$ je
	\begin{align*}
  a &= \lim_{x \to \infty} \frac{f(x)}{x} = \lim_{x \to \infty} \frac{(x-1)^3}{(x-2)^2x} \\
  & \stackrel{\text{LP}}{=} \lim_{x \to \infty} \frac{3(x-1)^2}{(x-2)(3x-2)} \stackrel{\text{LP}}{=} \lim_{x \to \infty} \frac{6(x-1)}{3x-2+3(x-2)} \\
  &= \lim_{x \to \infty} \frac{6(x-1)}{6x-8} = 1.
	\end{align*}
	Obdobně i pro $-\infty$.
	\begin{align*}
  b &= \lim_{x \to \infty} \frac{(x-1)^3}{(x-2)^2}-x = \lim_{x \to \infty} \frac{(x-1)^3 - x(x-2)^2}{(x-2)^2} \\
  &= \lim_{x \to \infty} \frac{x^2-x-1}{(x-2)^2} = \stackrel{\text{LP}}{=} \lim_{x \to \infty} \frac{2x-1}{2x-4} = 1.
	\end{align*}
	Obdobně i pro $-\infty$. Asymptota se směrnicí je tedy přímka $y=x+1$.
	\item Graf je v obrázku \ref{obrazekprubeh}.
\end{enumerate}
