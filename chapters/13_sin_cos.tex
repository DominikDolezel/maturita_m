\begin{definition}
    \textbf{Orientovaným úhlem} $\sphericalangle AVB$ nazýváme úhel s uspořádanou dvojicí
    polopřímek $\overrightarrow{VA}, \overrightarrow{VB}$ se společným počátkem v bodě $V.$
    Polopřímku $\overrightarrow{VA}$, resp. $\overrightarrow{VB}$ nazýváme \textbf{počátečním},
    resp. \textbf{koncovým ramenem}, bod $V$ \textbf{vrcholem}.
\end{definition}

\begin{definition}
    Nechť $\sphericalangle AVB$ je orientovaný úhel o základní velikosti $|\sphericalangle AVB| = \alpha$ dle def. \ref{raddef}. Pak
    \textbf{velikostí orientovaného úhlu} $\sphericalangle AVB$ rozumíme každé z
    čísel $\alpha +2k\pi, k \in \mathbb Z.$
\end{definition}

\begin{definition}
  Nechť je dána jednotková kružnice. Hodnotou funkce \textbf{sinus} (resp. \textbf{kosinus})
  v bodě $x$ nazveme $y$-ovou (resp. $x$-ovou) souřadnici průsečíku této kružnice s ramenem
  úhlu svírajícím s $x$-ovou osou úhel $\theta = x$.
\end{definition}

\begin{veta}
    Vlastnosti funkce sinus $f\colon y = \sin x$, nechť $k\in \mathbb Z$:
    \begin{enumerate}[$i.$]
        \item $\mathcal D(f)= \mathbb R$, $\mathcal H(f)= \left<  -1,1 \right> $;
       	\item $f$ je lichá;
        \item $f$ je rostoucí v intervalech $\left<  -\frac{\pi}{2}+2k\pi, \frac{\pi}{2}+2k\pi \right> $ a klesající v intervalech $\left<  \frac{\pi}{2}+2k\pi, \frac{3\pi}{2}+2k\pi \right> $;
        \item $f$ je omezená;
        \item $f$ má maximum v bodech $\frac{\pi}{2}+2k\pi$, minimum v bodech $\frac{3\pi}{2}+2k\pi.$
        \item $f$ je periodická s nejmenší periodou $2\pi$.
    \end{enumerate}
    Vlastnosti funkce kosinus $f\colon y = \cos x$, nechť $k \in \mathbb Z$:
    \begin{enumerate}[$i.$]
        \item $\mathcal D(f)= \mathbb R$, $\mathcal H(f)= \left<  -1,1 \right> $;
       	\item $f$ je sudá;
        \item $f$ je rostoucí v intervalech $\left<  \pi+2k\pi, 2\pi+2k\pi \right> $ a klesající v intervalech $\left<  0+2k\pi, \pi+2k\pi \right> $;
        \item $f$ je omezená;
        \item $f$ má maximum v bodech $2k\pi$, minimum v bodech $\pi+2k\pi$;
        \item $f$ je periodická s nejmenší periodou $2\pi$.
    \end{enumerate}
\end{veta}

\begin{priklad}
Určete $\sin \frac{5\pi}{3}.$
\end{priklad}

\begin{reseni}
Počítejme:
$$\sin \frac{5\pi}{3}=\sin \left ( 2\pi-\frac{\pi}{3} \right ) =\sin \left ( -\frac{\pi}{3} \right ). $$
\end{reseni}

\begin{priklad}
Načrtněte graf funkce $y=\cos (2x-\frac{\pi}{3}).$
\end{priklad}

\begin{reseni}
Počítejme:
$$
    y=\cos \left( 2x- \frac{\pi }{3} \right ) = \cos \left( 2 \left( x - \frac{\pi }{6} \right )  \right )  ,
$$
a tedy funkce je dvakrát huštěná a postunutá o $\frac{\pi }{6}$ doprava, jak je vidět v obrázku \ref{fig:fce2x-pi3}.
\end{reseni}

\begin{definition}
Funkce inverzní k funkci $f\colon y=\sin x$, kde $\mathcal D(f)=\left<  -\frac{\pi}{2},\frac{\pi}{2} \right>$, označená $f^{-1}\colon y=\arcsin x$, se nazývá funkce \textbf{arkussinus}.
\end{definition}
\begin{pozn}
    Funkce arkussinus má následující vlastnosti:
    \begin{enumerate}
    \item $\mathcal D(f^{-1}) = \left<  -1, 1 \right>$, $\mathcal H(f^{-1}) = \left<  -\frac{\pi}{2},\frac{\pi}{2} \right>$,
    \item $f^{-1}$ je rostoucí v celém $\mathcal D(f^{-1})$.
    \end{enumerate}
\end{pozn}


\begin{definition}
Funkce inverzní k funkci $g\colon y=\cos x$, kde $\mathcal D(g)=\left<  0,\pi \right>$, označená $g^{-1}\colon y=\arccos x$, se nazývá funkce \textbf{arkuskosinus}.
\end{definition}

\begin{pozn}
Funkce arkuskosinus má následující vlastnosti:
\begin{enumerate}
    \item $\mathcal D(g^{-1}) = \left<  -1, 1 \right>$, $\mathcal H(g^{-1}) = \left<  0, \pi \right>$,
    \item $g^{-1}$ je klesající v celém $\mathcal D(g^{-1})$.
\end{enumerate}
\end{pozn}

\begin{veta}[Trigonometrická jednička]
  Pro všechna $ x \in \mathbb{R} $ platí $\sin^2 x + \cos^2 x = 1$.
\end{veta}

\begin{veta}
    V pravoúhlém trojúhelníku $ABC$ s přeponou $AB$ platí:
    \begin{align*}
        \sin \alpha = \frac{ |BC| }{ |AB| }, & & \cos \alpha = \frac{|AC|}{|AB|}.
    \end{align*}
\end{veta}

\begin{pozn}
    \textbf{Goniometrická (ne)rovnice} je (ne)rovnice, v níž se vyskytují goniometrické funkce.
\end{pozn}
