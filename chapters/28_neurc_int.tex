\section{Neurčitý integrál}
\begin{definition}\label{integral}
Nechť $f(x)$ je definována na intervalu $I$. Funkce $F(x)$ se nazývá \textbf{primitivní}
k funkci $f(x)$ na $I$, jestliže platí $F^\prime(x)=f(x)$ pro každé $x\in I.$
Množina všech primitivních funkcí k funkci $f(x)$ na $I$ se nazývá \textbf{neurčitý integrál} funkce $f(x)$ a značí se $\int f(x)\d x$.
$$\int f(x)\d x= \left \{ F(x); F^\prime(x) = f(x), \forall x \in I\right \}. $$
\end{definition}

\begin{veta}
    Je-li $F(x)$ primitivní funkce k $f(x)$ na intervalu $I$, je také
    $F(x)+c$ primitivní funkce k $f(x)$ na $I$. Má-li funkce $f(x)$ alespoň
    jednu primitivní funkci, má jich nekonečně mnoho.
\end{veta}

\begin{veta}
Je-li funkce $f$ spojitá na intervalu $I$, pak na tomto intervalu existuje
primitivní funkce k funkci $f$.
\end{veta}

\begin{veta}[Pravidla pro integrování]
Nechť na intervalu $I$ existují integrály $\int f(x) \d x$ a~$\int g(x)\d x.$
Pak na $I$ existují také integrály $\int(f(x)\pm g(x))\d x$ a $\int \alpha f(x), \alpha \in \mathbb R,$ přičemž platí:
\begin{align*}
\int (f(x)\pm g(x)) \d x &= \int f(x)\d x \pm \int g(x) \d x,\\
\int \alpha f(x)\d x &= \alpha\int f(x)\d x.
\end{align*}
\end{veta}

\begin{pozn}[Tabulkové integrály]
\begin{align*}
    &\int 0\d x = c & & \, \\
    &\int \d x = x+c & & \, \\
    &\int x^n \d x = \frac{x^{n+1}}{n+1}+c, n\ne -1 & & \, \\
    &\int \frac{1}{x} \d x = \ln |x| + c & & \int \frac{1}{x+a}\d x = \ln |x+a|+c \\
    &\int e^x\d x = e^x+c & & \int e^{ax}\d x = \frac{1}{a}e^{ax}+c \\
    &\int a^x \d x = \frac{a^x}{\ln a} + c, a >0 & & \int a^{bx} \d x = \frac{1}{b}\cdot \frac{a^{bx}}{\ln a}+c, a>0\\
    &\int \sin x \d x = -\cos x + c & & \int \sin ax \d x = -\frac{1}{a}\cos ax+c \\
    &\int \cos x \d x = \sin x +c & & \int \cos ax\d x = \frac{1}{a}\sin ax + c \\
    &\int \frac{1}{\cos^2 x}\d x = \tg x + c & & \int \frac{1}{\cos^2 ax}\d x=\frac{1}{a}\tg ax+c\\
    &\int \frac{1}{\sin^2 x} \d x = -\cotg x+c & & \int \frac{1}{\sin^2 ax}\d x = \frac{1}{a}\cotg ax+c \\
    &\int \frac{1}{\sqrt{1-x^2} }\d x = \arcsin x + c & & \int \frac{1}{\sqrt{1-a^2x^2} }\d x = \frac{1}{a}\arcsin ax+c \\
    &\int \frac{1}{x^2+1}\d x = \arctg x + c & & \int \frac{1}{a^2x^2 + 1}\d x = \frac{1}{a}\arctg ax + c
\end{align*}
\end{pozn}

\begin{priklad}
Řešte integrál $\int x \d x.$
\end{priklad}

\begin{reseni}
Je $\int x \d x=\frac{x^2}{2}+c.$
\end{reseni}

\begin{veta}[Integrační metoda \textit{per partes}]
Nechť funkce $u(x) $ a $v(x)$ mají derivaci na intervalu $I$. Pak platí
$$\int u(x)v^\prime (x) \d x = u(x) v(x) - \int u^\prime (x) v(x) \d x,$$
pokud alespoň jeden z integrálů existuje.
\end{veta}

\begin{priklad}
Řešte $\int x \sin x \d x $.
\end{priklad}

\begin{reseni}
Řešme \textit{per partes}:
\begin{align*}
\int x \sin x \d x &=\left |\begin{array}{l l}
    u=x & u^\prime= 1\\
    v^\prime = \sin x & v=-\cos x
\end{array} \right | = x(-\cos x)-\int 1\cdot(-\cos x)\d x \\
&= -x\cos x+\int \cos x \d x=-x \cos x +\sin x+c
\end{align*}
\end{reseni}

\begin{priklad}
Řešte $\int (x^2+1)e^{-x}\d x.$
\end{priklad}

\begin{reseni}
Řešme \textit{per partes}:
\begin{align*}
    \int \left ( x^2+1 \right )e^{-x}\d x &= \left |\begin{array}{l l}
        u=x^2+1 & u^\prime= 2x\\
        v^\prime = e^{-x} & v=-e^{-x}
    \end{array} \right | \\
    &= (x^2+1)(-e^{-x})-\int 2x(-e^{-x})\d x = \dots
\end{align*}
\end{reseni}

\begin{pozn}[Substituční metoda]
    Označme si substituci $\varphi(x)=u.$ Tuto rovnost diferencujeme.
    $$\int f(\varphi(x))\varphi^\prime(x)\d x=\begin{vmatrix}
        \varphi(x)=u \\
        \varphi^\prime(x) \d x=du
    \end{vmatrix}=\int f(u)\d u.$$
\end{pozn}

\begin{priklad}
Řešte $\int \frac{(1+\ln x)^4}{x}\d x.$
\end{priklad}

\begin{reseni}
Metodou substituční:
\begin{align*}
    \int \frac{(1+\ln x)^4}{x}\d x &= \begin{vmatrix}
        1+\ln x=u \\
        \frac{1}{x}\d x=du
    \end{vmatrix}=\int u^4\d u \\
    &=\frac{u^5}{5}+c=\frac{(1+\ln x)^5}{5}+c
\end{align*}
\end{reseni}

\begin{priklad}
Řešte $\int e^{\sqrt{x} } \d x$.
\end{priklad}

\begin{reseni}
Řešíme substitucí $\sqrt{x}=t $. Počítejme
$$
    \int e^{\sqrt{x} } \d x = \begin{vmatrix}
    \sqrt{x} = t \\
    \frac{1}{2 \sqrt{x}} \d x = \d t
    \end{vmatrix} = \int e^t \cdot 2t \d t = 2 \left( te^t - \int e^t \d t \right ) = 2 e^t (t-1) + c.
$$
\end{reseni}

\begin{priklad}
Řešte $\int \sqrt{1-x^2} \d x.$
\end{priklad}

\begin{reseni}
Řešíme substitucí $x=\sin t$. Počítejme
$$
    \int \sqrt{1-x^2} \d x = \begin{vmatrix}
        x = \sin t \\
        \d x = \cos t \d t
    \end{vmatrix} = \int \sqrt{1 - \sin^2 t} \cos t \d t = \int \cos^2 t \d t.
$$
Nyní vyjdeme ze vzorce $\cos 2t = \cos^2 t  - \sin^2 t$ a trigonometrické jedničky, takže
$$
    \cos 2 t = \cos ^2 t  - \left( 1 - \cos ^2 t \right ) ,
$$
a tedy
$$
    \cos^2 t  = \frac{1+ \cos 2t}{2}.
$$
Můžeme tedy počítat
$$
    \int \cos^2 t \d t =\frac{t}{2} +\frac{1}{2}\int \cos 2 t \d t = \frac{t }{2} + \frac{1}{4} \sin 2t  + c .
$$
Ještě zbývá provést substituci zpět k $x$. Pokud $x = \sin t$, pak $t = \arcsin x$. Výsledek je tedy
$$
\frac{\arcsin t}{2} + \frac{1}{4} \sin (2 \arcsin t)+ c.
$$
\end{reseni}

\begin{pozn}[Integrály typu $\frac{f^\prime(x)}{f(x)}$]
    Platí
    $$\int \frac{f^\prime(x)}{f(x)}\d x=\begin{vmatrix}
        f(x)=t \\
        f^\prime(x) \d x = \d t
    \end{vmatrix}=\int \frac{dt}{t}=\ln |t|+c=\ln |f(x)|+c.$$
\end{pozn}

\begin{priklad}
Řešte $\int \tg x\d x.$
\end{priklad}

\begin{reseni}
Platí
$$\int \tg x\d x=\int \frac{\sin x}{\cos x}\d x.$$
Řešíme substitucí $\cos x=t.$ Počítejme
$$
    \int \frac{\sin x }{\cos x}\d x = \begin{vmatrix}
	\cos x = t \\
	-\sin x \d x = \d t
    \end{vmatrix} = \int \frac{-\d t }{t} = -\ln |t| + c = -\ln |\cos x| + c.
$$
\end{reseni}

\subsection*{Integrace racionálně lomených funkcí}
\begin{pozn}
    Každou racionální lomenou funkci $\frac{P(x)}{Q(x)}$ lze vyjádřit ve tvaru
    $$\frac{P(x)}{Q(x)}=S(x)+R_1(x)+\dots+R_n(x),$$
    kde $S(x)$ je polynom a $R_1(x),\dots,R_n(x)$ jsou parciální zlomky.
    Na libovolném intervalu, který neobsahuje kořeny $Q(x)$, jsou tyto funkce spojité,
    takže k nim existuje primitivní funkce a platí
    $$\int \frac{P(x)}{Q(x)} \d x=\int S(x)\d x + \int R_1(x)\d x + \dots + \int R_n(x)\d x.$$
\end{pozn}

\begin{pozn}[Integrace parciálních zlomků s reálnými kořeny ve jmenovateli]
    Parciální zlomky $\frac{A}{(x-\alpha)^k}$ integrujeme následovně:
    \begin{enumerate}[$i.$]
    \item je-li $k=1,$ pak triviálně
    $$\int \frac{A}{x-\alpha} \d x = A\ln |x-\alpha|+c.$$
   	\item je-li $k\geq 2,$ použijeme substituci:
    \begin{align*}
    \int \frac{A}{(x-\alpha)^k}\d x &=\begin{vmatrix}
      x-\alpha=t \\
      \d x = \d t
  \end{vmatrix}=A\int \frac{dt}{t^k}=A\int t^{-k}\d t \\
 & = A\cdot \frac{t^{-k+1}}{-k+1}+c=\frac{A}{(1-k)t^{k-1}}+c=\frac{A}{(1-k)(x-\alpha)^{k-1}}+c.
    \end{align*}
    \end{enumerate}
\end{pozn}

\begin{priklad}
Řešte $\int \frac{3x+16}{x^2-x-6}\d x$.
\end{priklad}

\begin{reseni}
Nejprve rozložíme zlomek $\frac{3x+16}{x^2-x-6}$ na praciální zlomky. Je
$$
    \frac{3x+16}{x^2-x-6} = \frac{3x + 16}{(x-2)(x+3)} = \frac{A }{x-2}+ \frac{B}{x+3}.
$$
Odsud dostáváme rovnici
$$
    3x + 16 = A(x+3) + B(x-2).
$$
Metodou porovnání koeficientů dostaneme soustavu
\begin{align*}
	x^1: & 3 = A+B, \\
	x^0: & 16 = 3A - 2B.
\end{align*}
Řešením této soustavy je $A = \frac{22}{4}$, $B = -\frac{7 }{5}$, a tedy
$$
    \int \frac{3x+16}{x^2-x-6}\d x = \frac{22}{4} \int \frac{1}{x-2} \d x -\frac{7 }{5} \int \frac{1}{x+3} \d x.
$$
Tyto integrály už spočteme jednoduše, je totiž
$$
    \int \frac{1}{x-2} \d x = \ln |x-2|+ c, \quad \int \frac{1}{x+3} \d x = \ln |x+ 3| + c.
$$
Výsledek je tedy
$$
    \int \frac{3x+16}{x^2-x-6}\d x = \frac{22}{4} \ln |x-2|-\frac{7 }{5}\ln |x+ 3|+c.
$$
\end{reseni}

\begin{pozn}[Integrace parciálních zlomků s komplexními kořeny ve jmenovateli]
Parciální zlomky $\frac{Mx+N}{(x^2+px+q)^n}$ integrujeme následovně:
\begin{enumerate}[1.]
    \item je-li $p=0$, pak $q>0,$ položíme $q=a^2,a>0$. Je tedy
    $$\frac{Mx+N}{(x^2+a^2)^n}.$$
    Pak
    $$\int \frac{Mx+N}{(x^2+a^2)^n}\d x = M\underbrace{\int \frac{x}{(x^2+a^2)^n}\d x}_{I.} + \underbrace{N\int \frac{1}{(x^2+a^2)^n}\d x}_{II.}.$$
    \begin{enumerate}[$I.$]
    \item Zlomky tvaru $\frac{x}{(a^2+a^2)^n}$ integrujeme následovně:
    \begin{enumerate}[$i.$]
    \item je-li $n=1$, pak
    \begin{align*}
        \int \frac{x}{x^2+a^2}\d x &= \frac{1}{2}\int \frac{2x}{x^2+a^2}\d x = \begin{vmatrix}
            x^2+a^2=t \\
            2x\d x = \d t
        \end{vmatrix} \\
        &=\frac{1}{2} \int \frac{\d t}{t}=\frac{1}{2}\ln |t|+c=\frac{1}{2}\ln(x^2+a^2)+c.
    \end{align*}
   	\item je-li $n \geq 2,$ pak
    \begin{align*}
        \int \frac{x}{(x^2+a^2)^n}\d x &= \begin{vmatrix}
            x^2+a^2=u \\
            2x\d x = \d u \\
            x\d x = \frac{1}{2}\d u
        \end{vmatrix}=\int \frac{1}{u^n}\cdot \frac{\d u}{2} \\
        &= \frac{1}{2}\int u^{-n}\d u = \frac{1}{2}\cdot \frac{u^{-n+1}}{-n+1}+c =\frac{1}{2(1-n)}\cdot \frac{1}{u^{n-1}}+c \\
        & = \frac{1}{2(1-n) }\cdot\frac{1}{(x^2+a^2)^{n-1}}+c.
    \end{align*}
    \end{enumerate}
   	\item Označme
    $$J_n(x,a)=\int \frac{1}{(x^2+a^2)^n}\d x.$$ Pak
    $$
        J_1(x)= \int \frac{1}{x^2+a^2} \d x = \frac{1}{a^2}\int \frac{1}{\left ( \frac{x}{a} \right )^2+1 }\d x =\frac{1}{a^2}\cdot \frac{1}{\frac{1}{a}}\cdot \arctg \frac{x}{a} + c = \frac{1}{a}\cdot \arctg \frac{x}{a}+c.
    $$
    $J_2$ určíme rekurentně tak, že vyjádříme metodou \textit{per partes} $J_1$ ($u=J_1$, $v^\prime =1$),
    $J_3$ tak, že vyjádříme metodou \textit{per partes} $J_2$, \dots
    \end{enumerate}
   	\item je-li $p\ne 0,$ upravíme zlomek tak, aby v čitateli byla derivace jmenovatele, tj.
    $$\frac{Mx+N}{(x^2+px+q)^n}=r\frac{2x+p}{(x^2+px+q)^n}+s \frac{1}{(x^2+px+q)^n}.$$
    Bude tedy
    $$\int \frac{Mx+N}{(x^2+px+q)^n}\d x =r\underbrace{\int \frac{2x+p}{(x^2+px+q)^n}\d x}_{I.}+s \underbrace{\int \frac{1}{(x^2+px+q)^n}\d x}_{II.}.$$
    \begin{enumerate}[$I.$]
    \item Integrál $\int \frac{2x+p}{(x^2+px+q)^n}\d x$ spočítáme obdobně jako první integrál v 1.
    \begin{enumerate}[$i.$]
    \item pro $n=1$ je
    $$\int \frac{2x+p}{x^2+px+q}\d x = \ln(x^2+px+q)+c.$$
   	\item pro $n \geq 2$ použijeme substituci
    \begin{align*}
    \int \frac{2x+p}{(x^2+px+q)^n}\d x &= \begin{vmatrix}
x^2+px+q=u \\
(2x+p)\d x = \d u
\end{vmatrix}=\int \frac{du}{u^n}\\
& = \frac{1}{1-n}\cdot \frac{1}{u^{n-1}}+c = \frac{1}{1-n}\cdot \frac{1}{(x^2+px+q)^{n-1}}+c.
    \end{align*}
    \end{enumerate}
   	\item Integrál $\int \frac{1}{(x^2+px+q)^n}\d x$ převedeme substitucí na výpočet
    integrálu $J_n$.
    \begin{align*}
        \int \frac{1}{(x^2+px+q)^n}\d x &= \int \frac{1}{\left [ \left ( x+\frac{p}{2} \right )^2 +a^2 \right ]^n }\d x = \begin{vmatrix}
            x+\frac{p}{2}=u \\
            \d x = \d u
        \end{vmatrix} \\
        &= \int \frac{1}{(u^2+a^2)^n}\d u  = J_n(u,a).
    \end{align*}
    \end{enumerate}
\end{enumerate}
\end{pozn}

\begin{priklad}
Řešte $\int \frac{1}{(x^2+2)^2}\d x = J_2.$
\end{priklad}

\begin{reseni}
Vyjádříme si \textit{per partes} $J_1$ tak, že derivujeme celý výraz $J_1$ a integrujeme
1. Dostaneme
$$
    \int \frac{1}{x^2 + 2} \d x = \frac{x }{x^2 + 2} + \int \frac{2x ^2 +2 -2}{\left( x^2 + 2 \right )^2  } \d x = \frac{x }{x^2 + 2} + 2 \underbrace{\int \frac{1}{x^2+2}\d x}_{J_1} + 4 \underbrace{\int \frac{1}{\left( x^2 + 2 \right )^2 }\d x}_{J_2}.
$$
Máme tedy rovnici
$$
    J_1 = \frac{x }{x^2 + 2}+ 2J_1 +4J_2,
$$
odkud dostáváme, že
$$
    J_2 = \frac{1}{4} \left( \frac{-x }{x^2 + 2} -J_1 \right ) .
$$
Integrál $J_1$ spočteme snadno, neboť
$$
    \int \frac{1}{x^2+2}\d x = \frac{1}{2} \int \frac{1}{\left( \frac{1}{\sqrt{2}} \right )^2 + 1 } \d x = \frac{1}{2} \cdot \sqrt{2} \arctg \left( \frac{1}{\sqrt{2}}\cdot  x \right )+ c.
$$
Celkem tedy máme, že
$$
    J_2 = \frac{1}{4} \left( \frac{-x }{x^2 + 2} -\frac{1}{\sqrt{2}} \cdot \arctg \left( \frac{1}{\sqrt{2}} \cdot x \right ) \right ) +c .
$$
\end{reseni}

\begin{priklad}
Řešte $\int \frac{5x+6}{x^2+28x+300}\d x.$
\end{priklad}

\begin{reseni}
Do čitatele chceme dostat derivaci jmenovatele. Máme
$$
    \int \frac{5x+6}{x^2+28x+300}\d x = \frac{5}{2} \int \frac{2x + 28}{x^2 + 28x + 300}\d x -64 \int  \frac{1}{x^2 + 28 x + 300} \d x.
$$
První integrál spočteme snadno, je totiž
$$
    \int \frac{2x + 28}{x^2 + 28x + 300}\d x = \ln |x^2 + 28x + 300| + c.
$$
Druhý integrál vede na $\arctg$. Počítejme
\begin{align*}
	\int  \frac{1}{x^2 + 28 x + 300} \d x &= \int  \frac{1}{(x+14)^2 + 104} \d x = \frac{1}{104} \int \frac{1}{\left( \frac{x + 14}{\sqrt{104}} \right )^2 + 1 } \d x \\
	&= \frac{\sqrt{104}}{104} \cdot \arctg \left( \frac{x+14}{\sqrt{104}} \right ) + c.
\end{align*}
Celkem tedy máme, že
$$
    \int \frac{5x+6}{x^2+28x+300}\d x = \frac{5}{2} \cdot \ln |x^2 + 28x + 300| - \frac{64}{\sqrt{104}} \cdot \arctg \left( \frac{x+14}{\sqrt{104}} \right ) + c.
$$
\end{reseni}


\subsection*{Integrace goniometrických funkcí}
\begin{pozn}
    Budeme vyšetřovat integrály typu
    $$\int R(\sin x, \cos x) \d x,$$
    kde $R(\sin x, \cos x)$ je funkce v proměnných $\sin x, \cos x.$ Substitucí
    $\tg \frac{x}{2}=t$ lze tento integrál vždy převést na integrál z racionální
    lomené funkce v proměnné $t$. \\
    Je-li $\tg \frac{x}{2}=t,$ pak ze vztahů v pravoúhlém trojúhelníku dostáváme
    \begin{align*}
        \sin \frac{x}{2}=\frac{t}{\sqrt{1+t^2} }, & & \cos \frac{x}{2}=\frac{1}{\sqrt{1+t^2} }.
    \end{align*}
    Dále
    \begin{align*}
        \sin x = 2\sin \frac{x}{2}\cos \frac{x}{2}=\frac{2t}{1+t^2}, & &
        \cos x = \cos^2 \frac{x}{2}-\sin^2 \frac{x}{2}=\frac{1-t^2}{1+t^2}.
    \end{align*}
    Pokud $\tg \frac{x}{2}=t,$ pak $\frac{x}{2}=\arctg t,$ a tedy
    $$\d x = \frac{2}{t^2+1}\d t.$$
\end{pozn}

\begin{priklad}
Řešte $\int \frac{\d x}{1-\cos x+\sin x}$.
\end{priklad}

\begin{reseni}
Řešíme substitucí $\tg \frac{x}{2}=t$. Jak je vidět v obrázku \ref{subs}, když $\tg \frac{x}{2} = t$, pak
jednotlivé odvěsny mají délky $t$, resp. $1$  a přepona má délku $\sqrt{1+t^2}$. Potom je
$$
    \sin \frac{x}{2} = \frac{t}{\sqrt{t^2 + 1}}, \quad \cos \frac{x }{2} = \frac{1}{\sqrt{t^2 + 1}}.
$$

Funkce $\sin x$ a $\cos x$ vyjádříme pomocí vzorců pro dvojnásobný argument:
\begin{align*}
	\sin x &= 2 \sin \frac{x}{2}\cos \frac{x }{2} = 2 \cdot \frac{t }{\sqrt{t^2 + 1}} \cdot \frac{1}{\sqrt{t^2 + 1}} = \frac{2t}{t^2 + 1}, \\
	\cos x &= \cos^2 \frac{x}{2} - \sin^2 \frac{x }{2} = \frac{1}{t^2 + 1} - \frac{t^2}{t^2 + 1} = \frac{1 - t^2}{t^2 + 1}.
\end{align*}
Potom máme
$$
    \int \frac{\d x}{1-\cos x+\sin x} = \begin{vmatrix}
	\tg \frac{x}{2} = t \\
	\frac{1}{\cos^2 \frac{x }{2}} \d x = \d t\\
	\d x = \cos^2 \frac{x }{2} \cdot \d t
    \end{vmatrix} = \int \frac{1}{1 - \frac{1-t^2}{t^2 + 1} + \frac{2t}{t^2+1}} \cdot \frac{1}{t^2 + 1}\d t.
$$
Tento výraz upravíme a dostaneme
$$
    \int \frac{\d x}{1-\cos x+\sin x} = \int \frac{1}{\frac{t^2 + 1-1+t^2 +2t}{t^2+1}} \cdot \frac{1}{t^2+1} \d t = \frac{1}{2}\int \frac{1}{t(t+1)}\d t.
$$
Tento zlomek rozložíme na parciální zlomky. Má být
$$
\frac{1}{t(t+1)} = \frac{A}{t} + \frac{B }{t+1},
$$
a tedy
$$
    1 = A(t+1) + Bt,
$$
odkud dostáváme, že
$$
    \frac{1}{t(t+1)} = \frac{1}{t} - \frac{1}{t+1}.
$$
Celkem tedy máme
$$
    \int \frac{\d x}{1-\cos x+\sin x} = \frac{1}{2} \ln |t| - \frac{1}{2} \ln |t+1| + c.
$$
Nakonec provedeme zpětnou substituci. Je
$$
    \int \frac{\d x}{1-\cos x+\sin x} = \frac{1}{2} \ln |\tg \frac{x}{2}| - \frac{1}{2} \ln |\tg \frac{x}{2}+1| +c.
$$
\end{reseni}

\begin{veta}
    Je-li funkce $R(\sin x, \cos x)$ lichá vzhledem k
    proměnné $\sin x$, používáme substituci $\cos x=t$.
\end{veta}

\begin{veta}
    Je-li funkce $R(\sin x, \cos x)$ lichá vzhledem k
    proměnné $\cos x$, používáme substituci $\sin x=t$.
\end{veta}

\begin{veta}
    Je-li funkce $R(\sin x, \cos x)$ sudá vzhledem k
    oběma proměnným, používáme substituci $\tg x=t$.
\end{veta}

\begin{priklad}
Řešte $\int \frac{2\cos x - \sin x}{\cos x - 2\sin x}\d x$.
\end{priklad}

\begin{reseni}
Řešíme substitucí $\tg x = t$. Jak je vidět z obrázku \ref{subs x}, pak je
$$
    \cos x = \frac{1}{\sqrt{t^2 + 1}}, \quad \sin x = \frac{t }{\sqrt{t^2 + 1}}.
$$
\begin{figure}[h]
	\centering
	\begin{tikzpicture}[scale=1.3]
    % Definice vrcholů
    \coordinate (A) at (0,0);
    \coordinate (B) at (4,0);
    \coordinate (C) at (4,3);

    % Vykreslení trojúhelníku
    \draw[thick] (A) -- (B) -- (C) -- cycle;

    % Značka pravého úhlu
    \draw (3.7,0) -- (3.7,0.3) -- (4,0.3);

    % Popisy stran
    \node[below] at (2,0) {$1$};
    \node[right] at (4,1.5) {$t$};
    \node[above left] at (2,1.5) {$\sqrt{1+t^2}$};

    % Vyznačení úhlu x/2
    \draw (0.8,0) arc[start angle=0, end angle=36.87, radius=0.8];
    \node at (1.1,0.35) {$x$};
	\end{tikzpicture}
	\caption{Substituce $\tg x = t$}
	\label{subs x}
\end{figure}
Máme tedy
$$
    \int \frac{2\cos x - \sin x}{\cos x - 2\sin x}\d x = \begin{vmatrix}
	\tg x = t \\
	\frac{1}{\cos^2 x} \d x = \d t \\
	\d x = \cos ^2 x \d t
    \end{vmatrix} = \bigintss \frac{2 \cdot \frac{1}{\sqrt{t^2 -1}} - \frac{t }{\sqrt{t^2 -1}}}{\frac{1}{\sqrt{t^2 -1}}-2 \cdot \frac{t }{\sqrt{t^2 -1}}}\cdot \frac{1}{t^2 + 1}  \d t .
$$
Po úpravě dostaneme
$$
    \int \frac{2\cos x - \sin x}{\cos x - 2\sin x}\d  = \int \frac{2-t}{(1-2t)(t^2+1)} \d t.
$$
Tento zlomek rozložíme na parciální zlomky. Má být
$$
    \frac{2-t}{(1-2t)(t^2+1)} = \frac{At + B }{t^2 + 1} + \frac{C }{1-2t}.
$$
Vyřešením rovnice
$$
    2-t = (At + B)(1-2t) + C(t^2 + 1)
$$
metodou porovnání koeficientů dostaneme
$$
    \frac{2-t}{(1-2t)(t^2+1)} = -\frac{t }{t^2 + 1} + \frac{2 }{1-2t}.
$$
První integrál vyřešíme snadno, je totiž
$$
    -\int \frac{t }{t^2 + 1}\d t = - \frac{1}{2}\int \frac{2t }{t^2 + 1}\d t = -\frac{1}{2} \ln|t^2 + 1| + c .
$$
Druhý integrál také. Máme
$$
    \int \frac{2 }{1-2t}\d t = -\int \frac{-2 }{1-2t}\d t = -\ln |1-2t|+ c.
$$
Celkem je tedy
$$
    \int \frac{2\cos x - \sin x}{\cos x - 2\sin x}\d x = -\frac{1}{2} \ln|t^2 + 1| -\ln |1-2t|+ c.
$$
Po zpětné substituci dostaneme výsledek
$$
\int \frac{2\cos x - \sin x}{\cos x - 2\sin x}\d x = -\frac{1}{2} \ln|(\tg x)^2 + 1| -\ln |1-2\tg x|+ c.
$$
\end{reseni}

\begin{pozn}
    Integrály $\int \sin^2 x \d x, \int \cos^2 x \d x$ neřešíme substitucí
    $\tg x  = t,$ nýbrž pomocí vzorců pro poloviční argument, tedy
    \begin{align*}
        \int \sin^2 x \d x &= \int \frac{1-\cos 2x}{2}\d x = \frac{1}{2}x - \frac{1}{4} \sin 2x + c, \\
        \int \cos^2 x \d x &= \int \frac{1+\cos 2x}{2}\d x = \frac{1}{2}x + \frac{1}{4} \sin 2x + c.
    \end{align*}
\end{pozn}

\begin{priklad}\label{prcos}
Řešte $\int \cos^2 x \d x$.
\end{priklad}

\begin{reseni}
Vyjdeme ze vzorce $\cos 2x = \cos^2 x  - \sin^2 x$ a trigonometrické jedničky $\sin^2 x + \cos^2 x =1$. Máme
$$
    \cos 2 x = \cos ^2 x  - \underbrace{\left( 1 - \cos ^2 x \right )}_{\sin^2 x} ,
$$
a tedy
$$
    \cos^2 x  = \frac{1+ \cos 2x}{2}.
$$
Můžeme tedy počítat
$$
    \int \cos^2 x \d x =\frac{x}{2} +\frac{1}{2}\int \cos 2 x \d x = \frac{x }{2} + \frac{1}{4} \sin 2x  + c .
$$
\end{reseni}

\subsection*{Integrace některých iracionálních funkcí}
\begin{veta}
    Integrály tvaru
    $$\int R(x,\sqrt[s]{x}) \d x, \quad \textrm{kde } s \in \mathbb N, s \geq 2$$
    řešíme substitucí $x=t^s$.
\end{veta}

\begin{veta}
    Integrály tvaru
    $$\int S(x, \sqrt[s_1]{x}, \dots, \sqrt[s_k]{x}  ) \d x ,$$
    kde $k \in \mathbb N, s_i \geq 2$, řešíme substitucí $x=t^s,$ kde $s$ je nejmenší společný násobek $s_1,\dots, s_k$.
\end{veta}

\begin{veta}
    Integrály tvaru
    $$\int R(x, \sqrt[s]{ax+b} )\d x, \quad \textrm{kde } s \in \mathbb N, s \geq 2, a,b \in \mathbb R$$
    řešíme substitucí $t^s=ax+b$.
\end{veta}

\begin{veta}
    Integrály tvaru
    $$\int R\left (x, \sqrt[s]{\frac{ax+b}{cx+d}} \right)\d x, \quad \textrm{kde } s \in \mathbb N, s \geq 2, a,b,c,d \in \mathbb R, ad-bc \ne 0$$
    řešíme substitucí $t^s=\frac{ax+b}{cx+d}.$
\end{veta}

\begin{veta}[Eulerovy substituce]
    Integrály tvaru
    $$\int R(x,\sqrt{ax^2+bx+c}) \d x, \quad \textrm{kde } a,b,c\in \mathbb R, a \ne 0$$
    řešíme substitucemi
    \begin{align*}
    \sqrt{ax^2+bx+c} &= \sqrt{|a|}\cdot t(x-\alpha) & & \textrm{je-li }D >0,  \\
    \sqrt{ax^2+bx+c} &= \pm\sqrt{|a|}\cdot x\pm t & & \textrm{je-li }D <0, a>0, \\
    \sqrt{ax^2+bx+c} &= \pm xt \pm \sqrt{c}  & & \textrm{je-li }c >0.
    \end{align*}
    Pokud $b=0$, řešíme integrály
    \begin{align*}
        &\int R(x, \sqrt{k^2-x^2} )\d x & & \textrm{substitucí } x=k\sin t, \\
        &\int R(x, \sqrt{x^2+k^2})\d x & & \textrm{substitucí } x=k\tg t, \\
        &\int R(x, \sqrt{x^2-k^2} )\d x & & \textrm{substitucí } x=\frac{k}{\sin t}.
    \end{align*}
\end{veta}

\begin{priklad}\label{probsah}
Řešte $\int \sqrt{a^2-x^2}\d x. $
\end{priklad}

\begin{reseni}
	Použijeme substituci $x=a\sin t$. Počítejme
	$$
	    \int \sqrt{a^2-x^2}\d x = \begin{vmatrix}
						x=a\sin t\\
						\d x = a \cos t \d t
					\end{vmatrix} = \int \sqrt{a^2 - a^2 \sin^2 t} \cdot a \cos t \d t = a^2 \int \cos^2 t \d t.
	$$
	Tento integrál už jsme řešili v příkladu \ref{prcos}. Dostaneme
	$$
	    \int \sqrt{a^2-x^2}\d x = a^2 \left( \frac{t }{2} + \frac{1}{4} \sin 2t \right )   + c.
	$$
	Po zpětné substituci $t=\arcsin \left( \frac{x}{a} \right ) $ máme výsledek
	$$
	    \int \sqrt{a^2-x^2}\d x = a^2 \left( \frac{\arcsin \left( \frac{x}{a} \right ) }{2} + \frac{1}{4} \sin 2 \left( \arcsin \left( \frac{x}{a} \right ) \right )  \right )   + c.
	$$
\end{reseni}

\begin{priklad}
Řešte $\int \frac{\d x}{x \sqrt{x^2+2} }$.
\end{priklad}

\begin{reseni}
Řešíme substitucí $x=\sqrt{2}\tg t. $ Počítejme
\begin{align*}\int \frac{\d x}{x \sqrt{x^2+2} } &= \begin{vmatrix}
	x = \sqrt{2}\tg t \\
	\d x = \sqrt{2}\cdot \frac{1}{\cos^2 t} \d t
\end{vmatrix} = \int \frac{\sqrt{2}\cdot \frac{1}{\cos^2 t} \d t}{\sqrt{2} \tg t \cdot \sqrt{2 \tg^2 t + 2}} \\
&= \int \frac{\d t }{\cos^2 t \cdot \frac{\sin^2 t }{\cos^2 t } \cdot \sqrt{2 \left( \tg ^2 t +1 \right ) }}\\
&= \int \frac{\d t }{\sqrt{2}  \cdot \sqrt{\frac{\sin^2 t + \cos ^2 t }{\cos^2 t }}} = \frac{1}{\sqrt{2}} \int \cos t \d t \\
&= \frac{1 }{\sqrt{2}} \sin t + c.
\end{align*}
Nakonec provedeme zpětnou substituci. Je totiž $t=\arctg \frac{x }{\sqrt{2}}$. Celkem máme
$$
    \int \frac{\d x}{x \sqrt{x^2+2} } = \frac{1}{\sqrt{2}} \sin \left( \arctg \frac{x }{\sqrt{2}} \right ) +c.
$$
\end{reseni}

\begin{priklad}
Řešte $\int \frac{1}{x}\sqrt{x^2-1}\d x, x \in (1,\infty). $
\end{priklad}

\begin{reseni}
Řešíme substitucí $x=\frac{1}{\sin t}.$ Dostaneme
\begin{align*}
\int \frac{1}{x}\sqrt{x^2-1}\d x &= \begin{vmatrix}
	x = \frac{1}{\sin t} \\
	\d x = \frac{-1}{\sin^2 t}\cos t \d t
\end{vmatrix} = \int \sin t \cdot \sqrt{\frac{1}{\sin^2 t } - 1} \cdot \frac{-1 }{\sin^2 t} \cos t \d t\\
&= \int \sqrt{\frac{1-\sin^2 t }{\sin^2 t}} \cdot \frac{-1}{\sin t} \cdot \cos t \d t \\
&= \int \sqrt{\frac{\cos^2 t }{\sin^2 t}} \cdot \frac{-1}{\sin t } \cdot \cos t \d t \\
&= -\int \tg^2 t \d t.
\end{align*}
Tento integrál vyřešíme pomocí substituce $\tg t = u$. Máme
$$
    \int \tg^2 t \d t = \begin{vmatrix}
	\tg t = u\\
	\frac{1}{\cos^2 t} \d t = \d u
    \end{vmatrix}=\int u^2 \frac{1}{u^2 + 1} \d u = \int \frac{u^2 + 1}{u^2 + 1} \d u - \int \frac{1}{u^2 +1} \d u.
$$
První integrál je triviálně roven $u+ c$, druhý je podle vzorce roven $\arctg |u|+c$.
Celkem tedy máme, že
$$
    \int \frac{1}{x}\sqrt{x^2-1}\d x = -u + \arctg |u|+c.
$$
Po dosazení za $t$ máme
$$
    \int \frac{1}{x}\sqrt{x^2-1}\d x = -\tg t + \arctg |\tg t|+c
$$
a konečně po dosazení za $x$
$$
    \int \frac{1}{x}\sqrt{x^2-1}\d x = -\tg \arcsin \left( \frac{1}{x} \right )  + \arctg \left |\tg \arcsin\left( \frac{1}{x} \right ) \right|+c
$$
\end{reseni}
