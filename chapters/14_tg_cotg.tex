\section{Funkce tangens, kotangens, arkustangens, arkuskotangens}
\begin{definition}
  Funkcí \textbf{tangens} (resp. \textbf{kotangens}) nazýváme funkci danou vztahem
  \begin{align*}
    \tg x = \frac{\sin x}{\cos x}, & & \textrm{resp.} & & \cotg x = \frac{\cos x}{\sin x}.
  \end{align*}
\end{definition}

\begin{veta}
    Vlastnosti funkce tangens $f\colon y = \tg x$, nechť $k\in \mathbb Z$:
    \begin{enumerate}[$i.$]
        \item $\mathcal D(f)= \mathbb R\smallsetminus \bigcup\limits_k\left \{\frac{\pi}{2}+ k\pi\right \} $, $\mathcal H(f)= \mathbb R $;
       	\item $f$ je lichá;
        \item $f$ je rostoucí v intervalech $\left ( -\frac{\pi}{2}+k\pi, \frac{\pi}{2}+k\pi \right ) $;
        \item $f$ není omezená;
        \item $f$ nemá extrémy;
        \item $f$ je periodická s nejmenší periodou $\pi$.
    \end{enumerate}
    Vlastnosti funkce kotangens $f\colon y = \cotg x$, nechť $k \in \mathbb Z$:
    \begin{enumerate}[$i.$]
        \item $\mathcal D(f)= \mathbb R\smallsetminus \bigcup\limits_k \left \{ k\pi \right \} $, $\mathcal H(f)= \mathbb R $;
       	\item $f$ je lichá;
        \item $f$ je klesající intervalech $\left ( k\pi, (k+1)\pi \right ) $;
        \item $f$ není omezená;
        \item $f$ nemá extrémy;
        \item $f$ je periodická s nejmenší periodou $\pi.$
    \end{enumerate}
\end{veta}

\begin{priklad}
Vypočtěte $\tg \left ( -\frac{19}{6}\pi \right ) $.
\end{priklad}

\begin{reseni}
	Protože je funkce tangens $\pi$-periodická, máme
	$$
	    \tg \left ( -\frac{19}{6}\pi \right ) = \tg \left ( -\frac{1}{6}\pi \right ) = \frac{\sin \left ( -\frac{1}{6}\pi \right )}{\cos \left ( -\frac{1}{6}\pi \right )}.
	$$
	Nyní víme, že funkce sinus je sudá, a proto $\sin \left ( -\frac{1}{6}\pi \right ) = \sin\left ( \frac{1}{6}\pi \right )=\frac{1}{2}$.
	Naopak, funkce kosinus je lichá, a proto $\cos \left ( -\frac{1}{6}\pi \right ) = -\cos \left ( \frac{1}{6}\pi \right )=-\frac{\sqrt{3}}{2}$.
	Celkem tedy máme, že
	$$
	    \tg \left ( -\frac{19}{6}\pi \right )=\frac{\frac{1}{2}}{-\frac{\sqrt{3}}{2}}=-\frac{1}{\sqrt{3}}.
	$$
\end{reseni}

\begin{definition}
  Funkce inverzní k funkci $f\colon y=\tg x$, kde $\mathcal D(f)=\left ( -\frac{\pi}{2}, \frac{\pi}{2} \right )$, označená $f^{-1}\colon y=\arctg x$, se nazývá funkce \textbf{arkustangens}.
\end{definition}

\begin{pozn}
Funkce arkustangens má následující vlastnosti:
\begin{enumerate}
\item $\mathcal D(f^{-1}) = \mathbb{R}$, $\mathcal H(f^{-1}) = \left ( -\frac{\pi}{2}, \frac{\pi}{2} \right )$,
\item $f^{-1}$ je rostoucí v celém $\mathcal D(f^{-1})$.
\end{enumerate}
\end{pozn}

\begin{definition}
  Funkce inverzní k funkci $g\colon y=\cotg x$, kde $\mathcal D(g)=\left ( 0, \pi \right )$, označená $g^{-1}\colon y=\arccotg x$, se nazývá funkce \textbf{arkuskotangens}.
\end{definition}

\begin{pozn}
Funkce arkuskotangens má následující vlastnosti:
\begin{enumerate}
\item $\mathcal D(g^{-1}) = \mathbb{R}$, $\mathcal H(g^{-1}) = \left ( 0, \pi \right )$,
\item $g^{-1}$ je klesající v celém $\mathcal D(g^{-1})$.
\end{enumerate}
\end{pozn}

\begin{veta}
    V pravoúhlém trojúhelníku $ABC$ s přeponou $AB$ platí:
    \begin{align*}
        \tg \alpha = \frac{ |BC| }{ |AC| }, & & \cotg \alpha = \frac{|AB|}{|BC|}.
    \end{align*}
\end{veta}
